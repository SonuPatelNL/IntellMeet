import { BrowserRouter, Routes, Route } from 'react-router-dom'
import MeetingRoom from './MeetingRoom'

function HomePage() {
  const handleCreateMeeting = async () => {
    const res = await fetch('http://localhost:5000/create-meeting', {
      method: 'POST',
    })
    const data = await res.json()
    alert(`Meeting Created!\nLink: ${data.link}`)
  }

  return (
    <div className="p-10 text-center bg-blue-900 min-h-screen text-white">
      <h1 className="text-4xl font-bold">IntellMeet 🚀</h1>
      <p>Server is running V2</p>
      <button 
        onClick={handleCreateMeeting}
        className="mt-6 px-6 py-3 bg-blue-600 text-white rounded-lg"
      >
        + Create New Meeting
      </button>
    </div>
  )
}

function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/" element={<HomePage />} />
        <Route path="/meet/:id" element={<MeetingRoom />} />
      </Routes>
    </BrowserRouter>
  )
}

export default App
